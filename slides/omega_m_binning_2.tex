\begin{frame}{How much does $\partial C_\ell/\partial\om$ depend on the other parameters?}
\small
At the benchmark-1 point, $\partial C_\ell/\partial\om$ recomputed with each \lcdm\ parameter $\theta_i$ shifted by $\pm3\%$;
change measured with Planck-like noise weighting (33\,$\mu$K-arcmin, $7'$ beam, $f_{\rm sky}=0.57$).

\vspace{0.6em}
\centering
\begin{tabular}{@{}lrrrrc@{}}
\toprule
& \multicolumn{4}{c}{change in $\partial C_\ell/\partial\om$, $\theta_i$ shifted $\pm3\%$} & same change at a \\
\cmidrule(lr){2-5}
shifted $\theta_i$ & TT & TE & EE & $\phi\phi$ & $1\sigma$ shift (TT/TE/EE) \\
\midrule
$h$        & $10.5\%$ & $11.0\%$ & $8.9\%$ & $1.7\%$ & $\sim3\%$ \\
$\omega_b$ & $4.6\%$  & $4.5\%$  & $3.4\%$ & $1.3\%$ & $\sim1\%$ \\
$A_s$      & $2.8\%$  & $2.7\%$  & $2.8\%$ & $3.0\%$ & $\sim0.8\%$ \\
$n_s$      & $2.0\%$  & $1.6\%$  & $1.7\%$ & $5.3\%$ & $\sim0.3\%$ \\
$\tau$     & $0.6\%$  & $0.8\%$  & $0.9\%$ & $0.1\%$ & $\sim1.6\%$ \\
\bottomrule
\end{tabular}

\vspace{1.0em}
\raggedright
\takeaway{For a $1\sigma$ shift in any other parameter, $\partial C_\ell/\partial\om$ changes by at most $\sim3\%$ (from $h$). A $\pm3\%$ step is a much larger move than $1\sigma$ for $h$, $\omega_b$, $A_s$, $n_s$, and a smaller one for $\tau$, hence its larger $1\sigma$ value.}
\end{frame}

\begin{frame}{Estimating $\partial C_\ell/\partial\om$ at fixed $\theta_s$}
\small
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{CLASS, numerical} (benchmark): five-point stencil, $\epsilon=1.2\times10^{-3}$,
\[ \frac{\partial C_\ell}{\partial\om}\simeq\frac{C_\ell(-2\epsilon)-8C_\ell(-\epsilon)+8C_\ell(\epsilon)-C_\ell(2\epsilon)}{12\,\epsilon}, \]
with $h$ re-solved at every point so that CLASS's $100\theta_s = 1.0419693546$ (achieved to $4\times10^{-9}$).
High precision: $\ell_{\max}=3500$, refined $\ell$ and $q$ sampling, tight tolerances (${\sim}9$ min/run).

\vspace{0.5em}
\textbf{Emulators, analytic}: exact network Jacobian (no step), along the fixed-$\theta_s$ direction
\[ \frac{\partial C_\ell}{\partial\om}\Big|_{\theta_s}=\frac{\partial C_\ell}{\partial\omega_c}\Big|_h+\frac{\partial C_\ell}{\partial h}\Big|_{\omega_c}\frac{dh}{d\om}\Big|_{\theta_s},\quad \frac{dh}{d\om}\Big|_{\theta_s}=-3.524 \]
(slope from CLASS; $d\ln h/d\ln\om=-0.748$).
\column{0.43\textwidth}
\includegraphics[width=\textwidth]{dcl_spt_an_vs_num}

{\scriptsize Analytic $-$ numerical, same emulator, per unit $\om$, $/C_\ell$. With the exact slope: $\lesssim10^{-4}$ (Jacobians exact).
The pipeline's fitted slope ($-0.74844$ vs $-0.74773$) alone moves the \texttt{lcdm v1} derivative by up to 0.8\% (TT), 1.4\% (EE).}
\end{columns}
\vspace{0.2em}
{\scriptsize Fiducial: $\omega_b\,0.02237$, $\omega_c\,0.12$, $h\,0.674$, $\tau\,0.054$, $n_s\,0.965$, $\ln(10^{10}A_s)\,3.044$, $m_\nu$ 0.06\,eV, HMcode.}
\end{frame}

\begin{frame}{A smooth $\om(\ell)$: the parametrisation}
\small
Instead of $\om$ bins with sharp edges, a smooth $\ell$-dependence added to the spectra in linear response:
\[ C_\ell = C_\ell^{\Lambda{\rm CDM}}(\omega_b,h,\tau,n_s,A_s,\omega_{m0}) \;+\; \frac{\partial C_\ell}{\partial\om}\Big|_{\theta_s,\,\omega_b,\,\tau,\,n_s,\,A_s}\,\delta\om(\ell),
\qquad \delta\om(\ell)=a+b\,x+c\,x^2,\quad x=\frac{\ell-\ell_c}{1000}. \]
\vspace{-0.6em}
\begin{itemize}\small
  \item Same $\delta\om(\ell)$ on TT, TE, EE, $\phi\phi$, each with its own derivative. The derivative is the exact network Jacobian (\texttt{lcdm v1}) at the \emph{current} point,
        along the fixed-$\theta_s$ direction: $\partial_{\omega_c}C_\ell|_h+\partial_h C_\ell\,(dh/d\om)_{\theta_s}$, with $d\ln h/d\ln\om|_{\theta_s}$ from a fit to CLASS (good to 0.3\% over the prior box).
  \item $a$ is degenerate with $\omega_{m0}$ (plus the matching $h$ shift): $a\equiv0$, so $\omega_{m0}=\om(\ell_c)$ and $b$, $c$ carry all the $\ell$-dependence.
  \item $\ell_c=400$: where $\omega_{m0}$ and $b$ decorrelate (Fisher). Moving $\ell_c$ only reparametrises the same model.
  \item Fisher: $\sigma(b)=0.0041$, $\sigma(c)=0.0053$, $\sigma(\omega_{m0})=0.0013$. Priors $b,c\in[-0.05,0.05]$ (flat).
  \item $\sigma[\om(\ell)]$: $0.0014$ at $\ell\simeq200$--$400$, $0.0027$ at 1200, $0.005$ at 1600, $0.018$ at 2500: the data constrain the curve only below $\ell\sim1500$.
\end{itemize}
\vspace{0.2em}
\takeaway{Linear response is accurate near $b=c=0$ (error $\sqrt{\chi^2}=0.01$ for a $1\sigma$ shift, $0.1$ for $3\sigma$), i.e.\ for a null test.
Far out in the high-$\ell$ tails it is a phenomenological model, not ``$\om$ at that $\ell$''.}
\end{frame}
